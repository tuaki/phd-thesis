%%% Minor tweaks of style

% These macros employ a little dirty trick to convince LaTeX to typeset
% chapter headings sanely, without lots of empty space above them.
% Feel free to ignore.
\makeatletter
\def\@makechapterhead#1{
  {\parindent \z@ \raggedright \normalfont
   \Huge\bfseries \thechapter\quad #1
   \par\nobreak
   \vskip 20\p@
}}
\def\@makeschapterhead#1{
  {\parindent \z@ \raggedright \normalfont
   \Huge\bfseries #1
   \par\nobreak
   \vskip 20\p@
}}
\makeatother

\newcounter{autoFootnoteLabel}
\def\lastFootnoteLabel{}

% Usage: \labeledFootnote[<label>]{<text>} to create a footnote with a specific label. If the label is not provided, it will be generated automatically.
\NewDocumentCommand{\labeledFootnote}{O{} m}{%
  % Determine label
  \def\currentLabel{#1}%
  \ifx\currentLabel\empty
    \stepcounter{autoFootnoteLabel}%
    \edef\currentLabel{autofn:\arabic{autoFootnoteLabel}}%
  \fi
  % Remember for later \footref
  \global\let\lastFootnoteLabel\currentLabel
  % Create footnote
  \footnote{\label{\currentLabel}#2}%
}

% Usage: \footnoteRef[<label>] to create a reference to a footnote with the given label. If the label is not provided, it will refer to the last created `\labeledFootnote`.
\NewDocumentCommand{\footnoteRef}{O{}}{%
  \def\currentLabel{#1}%
  \ifx\currentLabel\empty
    \let\currentLabel\lastFootnoteLabel
  \fi
  \hyperref[\currentLabel]{\footnotemark[\getrefnumber{\currentLabel}]}%
}
